/** Concepts **/

Basic Code:
`{
template <typename X, typename Y>
concept Same = std::same_as<X, Y>;

template <typename T, typename... Ts>
concept OneOf = (Same<T, Ts> || ...);
}`

/* 32-bit int&float */
i32, f32 ...............................................= nx32

/* 64-bit int&float */
i64, f64 ...............................................= nx64

/* 32-bit+64-bit int&float */
i32, i64, f32, f64 .....................................= CommonNumber

/* Commonly used numbers and strings */
i32, i64, f32, f64, cstr, str ..........................= CommonTypes

/* All integer + unsiged + floating */
i8, i16, i32, i64, u8, u16, u32, u64, f32, f64, fld ....= Numbers

/* Type references */
`{
template <typename T>
concept Ref = OneOf<T, T&, const T&>;
}`
