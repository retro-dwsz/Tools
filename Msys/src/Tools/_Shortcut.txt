/*--------------------- Types Aliases ---------------------*/

/** Basic aliases **/ 

/* numbers */
i8 .....................................................= int8_t
i16 ....................................................= int16_t
i32 ....................................................= int32_t
i64 ....................................................= int64_t

u8 .....................................................= unt8_t
u16 ....................................................= unt16_t
u32 ....................................................= unt32_t
u64 ....................................................= unt64_t

f32 ....................................................= float 
f64 ....................................................= double
fld ....................................................= long double 

/* strings */
str ....................................................= std::string 
wstr ...................................................= std::wstring
str16 ..................................................= std::u16string
str32 ..................................................= std::u32string

strview ................................................= std::string_view
sstream ................................................= std::stringstream
ostream ................................................= std::ostringstream

/* containers */
ivec<T> ................................................= Tools::ivec<T>
vec<T> .................................................= std::vector<T>
arr<T, idx> ............................................= std::array<T, idx>
map<K, V> ..............................................= std::map<K, V> 
umap<K, V> .............................................= std::unordered_map<K, V> 
set<T> .................................................= std::set<T>
uset<T> ................................................= std::unordered_set<T> 
pair<K, V> .............................................= std::pair<K, V> 
initl<T> ...............................................= std::initializer_list<T>
span<T> ................................................= std::span<T>
tuple<T> ...............................................= std::tuple<T> 
list<T> ................................................= std::list<T> 

/* type containers */
topt<T>.................................................= std::optional<T>
tvar<T...> .............................................= std::varaint<T...>
texp<A, B> .............................................= std::expected<A, B> 

/* Pointer */
/* Pointer to data*/
ptr<T> .................................................= T*;

/* Pointer to constant data */
ptrcd<T> ...............................................= const T*;

/* Constant pointer to data */
cptr<T> ................................................= T const*;

/* Constant pointer to constant data*/
cptrcd<T> ..............................................= const T *const;


/** Concepts (or more like union of "accepted" types) **/

Basic Code:
`{
template <typename X, typename Y>
concept Same = std::same_as<X, Y>;

template <typename T, typename... Ts>
concept OneOf = (Same<T, Ts> || ...);
}`
